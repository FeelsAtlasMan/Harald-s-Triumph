#Reports from Macedonia

	id = 1122113

	fire_only_once = yes
	
	trigger = {
		tag = BYZ
		MCD = { exists = no }
		has_country_flag = macedonian_peasent_revolt
	}

		name = "A deserved punishment!"
		any_pop = { limit = { location = { OR = { is_core = MCD owned_by = THIS } } } reduce_pop = 0.96 }

	}
}
